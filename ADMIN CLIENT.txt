local function makePlayersPanel(parent)
	local left = Instance.new("Frame")
	left.Size = UDim2.new(0.5, -10, 1, 0)
	left.BackgroundTransparency = 1
	left.Parent = parent

	local lbl = Instance.new("TextLabel")
	lbl.Size = UDim2.new(1, -120, 0, 24)
	lbl.BackgroundTransparency = 1
	lbl.TextColor3 = COLORS.BLACK_TEXT
	lbl.Text = "PLAYERS"
	lbl.Font = Enum.Font.GothamBold
	lbl.TextSize = 14
	lbl.TextXAlignment = Enum.TextXAlignment.Left
	lbl.Parent = left

	local allBtn = Instance.new("TextButton")
	allBtn.Size = UDim2.new(0, 50, 0, 24)
	allBtn.Position = UDim2.new(1, -120, 0, 0)
	allBtn.BackgroundColor3 = COLORS.SELECT_BG
	allBtn.TextColor3 = COLORS.BLACK_TEXT
	allBtn.Text = "ALL"
	allBtn.Font = Enum.Font.GothamBold
	allBtn.TextSize = 12
	allBtn.BorderSizePixel = 0
	allBtn.Parent = left

	local refBtn = Instance.new("TextButton")
	refBtn.Size = UDim2.new(0, 60, 0, 24)
	refBtn.Position = UDim2.new(1, -60, 0, 0)
	refBtn.BackgroundColor3 = COLORS.SELECT_BG
	refBtn.TextColor3 = COLORS.BLACK_TEXT
	refBtn.Text = "↻ REF"
	refBtn.Font = Enum.Font.GothamBold
	refBtn.TextSize = 12
	refBtn.BorderSizePixel = 0
	refBtn.Parent = left

	local scroll = Instance.new("ScrollingFrame")
	scroll.Size = UDim2.new(1, 0, 1, -30)
	scroll.Position = UDim2.new(0, 0, 0, 30)
	scroll.BackgroundColor3 = COLORS.WHITE_BG
	scroll.BorderSizePixel = 1
	scroll.BorderColor3 = COLORS.BORDER
	scroll.ScrollBarThickness = 4
	scroll.Parent = left

	local pad = Instance.new("UIPadding")
	pad.PaddingTop = UDim.new(0, 4); pad.PaddingBottom = UDim.new(0, 4)
	pad.PaddingLeft = UDim.new(0, 4); pad.PaddingRight = UDim.new(0, 4)
	pad.Parent = scroll

	local selected: {[number]: boolean} = {}

	local function update()
		scroll:ClearAllChildren()
		local layout = Instance.new("UIListLayout")
		layout.Padding = UDim.new(0, 2); layout.SortOrder = Enum.SortOrder.LayoutOrder
		layout.Parent = scroll
		local i = 0
		for _, p in ipairs(Players:GetPlayers()) do
			i += 1
			local sel = selected[p.UserId] == true
			local b = Instance.new("TextButton")
			b.Size = UDim2.new(1, 0, 0, 28)
			b.BackgroundColor3 = sel and COLORS.SELECT_BG or COLORS.WHITE_BG
			b.TextColor3 = COLORS.BLACK_TEXT
			b.Text = "  " .. p.Name .. " [ID: " .. p.UserId .. "]" .. (p == player and " (You)" or "")
			b.TextXAlignment = Enum.TextXAlignment.Left
			b.Font = Enum.Font.Gotham
			b.TextSize = 13
			b.Name = tostring(p.UserId)
			b.LayoutOrder = i
			b.BorderSizePixel = 0
			b.Parent = scroll
			b.MouseButton1Click:Connect(function()
				if selected[p.UserId] then
					selected[p.UserId] = nil
					b.BackgroundColor3 = COLORS.WHITE_BG
				else
					selected[p.UserId] = true
					b.BackgroundColor3 = COLORS.SELECT_BG
				end
			end)
		end
		scroll.CanvasSize = UDim2.new(0, 0, 0, layout.AbsoluteContentSize.Y + 10)
	end

	allBtn.MouseButton1Click:Connect(function()
		for _, c in ipairs(scroll:GetChildren()) do
			if c:IsA("TextButton") then
				c.BackgroundColor3 = COLORS.SELECT_BG
				selected[tonumber(c.Name)] = true
			end
		end
	end)

	refBtn.MouseButton1Click:Connect(function()
		update()
		refBtn.Text = "✓ OK"; task.wait(0.8); refBtn.Text = "↻ REF"
	end)

	update()
	return selected, update
end

local function collectSelected(sel: {[number]: boolean}): {number}
	local out = {}
	for uid in pairs(sel) do table.insert(out, uid) end
	return out
end

local selectedPlaceId: number? = nil

local function renderPlaces()
	scrollPlaces:ClearAllChildren()
	local layout = Instance.new("UIListLayout")
	layout.Padding = UDim.new(0, 2); layout.SortOrder = Enum.SortOrder.LayoutOrder; layout.Parent = scrollPlaces
	local search = string.lower(searchBox.Text)
	local i = 0
	for _, place in ipairs(PLACES_LIST) do
		if search == "" or string.find(string.lower(place.Name), search, 1, true) then
			i += 1
			local b = Instance.new("TextButton")
			b.Size = UDim2.new(1, 0, 0, 28); b.BackgroundColor3 = (selectedPlaceId == place.ID) and COLORS.SELECT_BG or COLORS.WHITE_BG
			b.TextColor3 = COLORS.BLACK_TEXT; b.Text = "  " .. place.Name .. " [ID: " .. place.ID .. "]"
			b.TextXAlignment = Enum.TextXAlignment.Left; b.Font = Enum.Font.Gotham
			b.TextSize = 13; b.LayoutOrder = i; b.BorderSizePixel = 0; b.Parent = scrollPlaces
			b.MouseButton1Click:Connect(function()
				for _, c in ipairs(scrollPlaces:GetChildren()) do
					if c:IsA("TextButton") then c.BackgroundColor3 = COLORS.WHITE_BG end
				end
				b.BackgroundColor3 = COLORS.SELECT_BG; selectedPlaceId = place.ID
			end)
		end
	end
	scrollPlaces.CanvasSize = UDim2.new(0, 0, 0, layout.AbsoluteContentSize.Y + 10)
end

searchBox:GetPropertyChangedSignal("Text"):Connect(renderPlaces)

local function loadPlacesFromServer()
	refPlacesBtn.Text = "..."
	local ok, list = pcall(function() return requestRemote:InvokeServer("places") end)
	if ok and type(list) == "table" then
		PLACES_LIST = list; renderPlaces(); refPlacesBtn.Text = "✓ OK"
	else refPlacesBtn.Text = "ERR" end
	task.wait(1); refPlacesBtn.Text = "↻ REF"
end

refPlacesBtn.MouseButton1Click:Connect(loadPlacesFromServer)

local cam = workspace.CurrentCamera
local UIS = game:GetService("UserInputService")

	for _, v in ipairs(toolsFrame:GetDescendants()) do
		if v:IsA("TextBox") and v.Text == "30" then radiusBox = v end
		if v:IsA("TextBox") and v.Text == "60" then forceBox = v end
	end

local function shootLaser(isMinigun)
	local char = player.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if not root then return end

	local origin = cam.CFrame.Position
	local direction = cam.CFrame.LookVector

	local rayParams = RaycastParams.new()
	rayParams.FilterType = Enum.RaycastFilterType.Exclude
	rayParams.FilterDescendantsInstances = {char}
	local hit = workspace:Raycast(origin, direction * 400, rayParams)
	local endPos = hit and hit.Position or (origin + direction * 400)

	local beam = Instance.new("Part")
	beam.Anchored = true; beam.CanCollide = false; beam.CanQuery = false
	beam.Material = Enum.Material.Neon
	beam.Color = isMinigun and Color3.fromRGB(255, 150, 0) or Color3.fromRGB(255, 60, 60)
	beam.Size = Vector3.new(0.15, 0.15, (endPos - origin).Magnitude)
	beam.CFrame = CFrame.lookAt((origin + endPos) / 2, endPos)
	beam.Parent = workspace
	task.delay(0.12, function() if beam then beam:Destroy() end end)

	local rad = radiusBox and tonumber(radiusBox.Text) or 30
	local frc = forceBox and tonumber(forceBox.Text) or 60
	explodeRemote:FireServer(origin, direction, rad, frc)
end

local minigunLoop = false

local function onToolActivated(tool: Tool)
	if tool.Name == "Load Explode Tool" then
		shootLaser(false)
	elseif tool.Name == "Minigun Explode Tool" then
		if not minigunLoop then
			minigunLoop = true
			task.spawn(function()
				while minigunLoop and tool.Parent do
					shootLaser(true)
					task.wait(0.15)
				end
			end)
		end
	elseif tool.Name == "Freeze Tool" then
		local mouse = player:GetMouse()
		if mouse.Target then
			local hitChar = mouse.Target:FindFirstAncestorOfClass("Model")
			if hitChar then
				local hum = hitChar:FindFirstChildOfClass("Humanoid")
				if hum then
					local root = hitChar:FindFirstChild("HumanoidRootPart")
					if root then
						root.Anchored = not root.Anchored
						hum.WalkSpeed = root.Anchored and 0 or 16
					end
				end
			end
		end
	end
end

UIS.InputEnded:Connect(function(input, gpe)
	if not gpe and input.UserInputType == Enum.UserInputType.MouseButton1 then minigunLoop = false end
end)
UIS.TouchEnded:Connect(function() minigunLoop = false end)

local function hookTool(inst: Instance)
	if inst:IsA("Tool") then
		inst.Activated:Connect(function() onToolActivated(inst) end)
	end
end

local function hookBackpack(bp: Instance)
	for _, c in ipairs(bp:GetChildren()) do hookTool(c) end
	bp.ChildAdded:Connect(hookTool)
end

local function hookCharacter(char: Model)
	for _, c in ipairs(char:GetChildren()) do hookTool(c) end
	char.ChildAdded:Connect(hookTool)
	local bp = player:FindFirstChildOfClass("Backpack")
	if bp then hookBackpack(bp) end
end

player.CharacterAdded:Connect(hookCharacter)
if player.Character then hookCharacter(player.Character) end
local existingBp = player:FindFirstChildOfClass("Backpack")
if existingBp then hookBackpack(existingBp) end
