local function getRank(player: Player): number

local logQueue = {}
local lastLogSave = os.clock()
local jobId = game.JobId ~= "" and game.JobId or "STUDIO"

local function pushLog(admin: Player, action: string, details: string)
	local data = {
		adminName = admin.Name,
		adminId = admin.UserId,
		action = action,
		details = details,
		jobId = jobId,
		timestamp = os.time(),
		logId = HttpService:GenerateGUID(false)
	}
	table.insert(logQueue, data)
	warn(string.format("[LOG] %s (%d) выполнил %s: %s", admin.Name, admin.UserId, action, details))
end

local function saveLogsToDataStore()
	if #logQueue == 0 then return end
	local batch = logQueue
	logQueue = {}
	pcall(function()
		for _, data in ipairs(batch) do
			logStore:SetAsync("log_" .. data.logId, data)
		end
	end)
	lastLogSave = os.clock()
end

task.spawn(function()
	while true do
		task.wait(5)
		if #logQueue >= 20 or (os.clock() - lastLogSave > 30 and #logQueue > 0) then
			saveLogsToDataStore()
		end
	end
end)

game:BindToClose(function()
	saveLogsToDataStore()
	task.wait(2)
end)

-- ===== BAN DB =====
local function isBanned(userId: number): (boolean, string?)
	local ok, data = pcall(function() return banStore:GetAsync("ban_" .. userId) end)
	if ok and data then
		if data.untilTime and os.time() > data.untilTime then
			pcall(function() banStore:RemoveAsync("ban_" .. userId) end)
			return false, nil
		end
		return true, data.reason
	end
	return false, nil
end

local function setBan(userId: number, reason: string, durationHours: number)
	local untilTime = nil
	if durationHours > 0 then untilTime = os.time() + (durationHours * 3600) end
	pcall(function() banStore:SetAsync("ban_" .. userId, {reason = reason, at = os.time(), untilTime = untilTime}) end)
end

local function clearBan(userId: number)
	pcall(function() banStore:RemoveAsync("ban_" .. userId) end)
end

-- ===== HELPERS =====
local function fail(msg: string, ...) warn("[AdminServer] " .. msg, ...) end

local function checkCooldown(player: Player, action: string, rank: number): boolean
	if rank >= BAN_RANK then return true end
	local map = lastAction[action]
	if not map then map = {}; lastAction[action] = map end
	local now = os.clock()
	if now - (map[player.UserId] or 0) < (ACTION_COOLDOWN[action] or 0.5) then return false end
	map[player.UserId] = now
	return true
end

local function sanitizeText(s: any, maxLen: number): string
	if type(s) ~= "string" then return "" end
	s = string.gsub(s, "[%z\1-\31\127]", "")
	if #s > maxLen then s = string.sub(s, 1, maxLen) end
	return s
end

local function collectTargets(player: Player, ids: any): {Player}
	local out = {}
	if type(ids) ~= "table" then return out end
	for _, uid in ipairs(ids) do
		local p = Players:GetPlayerByUserId(uid)
		if p then table.insert(out, p) end
	end
	return out
end

-- ===== EXPLODE TOOLS =====
local function doExplode(origin: Vector3, direction: Vector3, requester: Player, radius: number, force: number)
	local rayParams = RaycastParams.new()
	rayParams.FilterType = Enum.RaycastFilterType.Exclude
	rayParams.FilterDescendantsInstances = {requester.Character or workspace}
	rayParams.IgnoreWater = true
	local result = workspace:Raycast(origin, direction.Unit * 500, rayParams)
	local hitPos = result and result.Position or (origin + direction.Unit * 500)

	local explosion = Instance.new("Explosion")
	explosion.Position = hitPos; explosion.BlastRadius = radius; explosion.BlastPressure = 0
	explosion.DestroyJointRadiusPercent = 0; explosion.ExplosionType = Enum.ExplosionType.NoCraters
	explosion.Parent = workspace

	for _, part in ipairs(workspace:GetPartBoundsInRadius(hitPos, radius)) do
		if part:IsA("BasePart") then
			if part.Anchored then part.Anchored = false end
			if part.CanSetNetworkOwnership() then pcall(function() part:SetNetworkOwner(nil) end) end
			local dir = (part.Position - hitPos)
			local dist = math.max(dir.Magnitude, 1)
			local falloff = math.clamp(1 - dist / radius, 0, 1)
			local impulse = dir.Unit * force * falloff + Vector3.new(0, force * 0.4 * falloff, 0)
			pcall(function() part:ApplyImpulse(impulse * part.AssemblyMass) end)
		end
	end
end

explodeRemote.OnServerEvent:Connect(function(player, origin, direction, radius, force)
	local rank = getRank(player)
	if rank < MOD_RANK then return end
	if rank < BAN_RANK then
		local now = os.clock()
		if now - (explodeLast[player.UserId] or 0) < 0.6 then return end
		explodeLast[player.UserId] = now
	end
	if typeof(origin) ~= "Vector3" or typeof(direction) ~= "Vector3" or direction.Magnitude < 0.01 then return end
	local char = player.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if root and (origin - root.Position).Magnitude > 20 then origin = root.Position end
	doExplode(origin, direction, player, math.clamp(tonumber(radius) or 30, 5, 150), math.clamp(tonumber(force) or 60, 10, 300))
end)

actionRemote.OnServerEvent:Connect(function(player, action, targetIds, payload)
	local rank = getRank(player)
	if rank < MIN_RANK or not checkCooldown(player, action, rank) then return end
	payload = type(payload) == "table" and payload or {}

	local modActions = {
		Hint = true, Message = true, Kick = true, Kill = true,
		F3X = true, ExplodeTool = true, Sword = true, Minigun = true,
		FreezeTool = true, Char = true
	}
	if modActions[action] and rank < MOD_RANK then return end

	local targets = collectTargets(player, targetIds)
	local tNames = {}
	for _, t in ipairs(targets) do table.insert(tNames, t.Name) end
	local targetStr = #tNames > 0 and table.concat(tNames, ", ") or "None"

	if action == "Hint" or action == "Message" then
		local text = sanitizeText(payload.text, 300)
		local dur  = math.clamp(tonumber(payload.duration) or 5, 1, 120)
		if text == "" then return end
		local isMessage = (action == "Message")
		for _, p in ipairs(Players:GetPlayers()) do
			local pg = p:FindFirstChildOfClass("PlayerGui") or p:WaitForChild("PlayerGui", 5)
			if pg then
				local obj = Instance.new(isMessage and "Message" or "Hint")
				obj.Text = isMessage and text or ("[GLOBAL ADMIN] " .. text)
				obj.Parent = pg
				task.delay(dur, function() if obj and obj.Parent then obj:Destroy() end end)
			end
		end
		pushLog(player, action, "Text: " .. text .. " (Dur: " .. dur .. ")")
		return
	end

	if action == "Kick" then
		local reason = sanitizeText(payload.text, 200)
		for _, target in ipairs(targets) do target:Kick("Кикнут админом. Причина: " .. (reason ~= "" and reason or "без причины")) end
		pushLog(player, "Kick", "Targets: " .. targetStr .. " | Reason: " .. (reason ~= "" and reason or "None"))
		return
	end

	if action == "Kill" then
		for _, target in ipairs(targets) do
			local hum = target.Character and target.Character:FindFirstChildOfClass("Humanoid")
			if hum then pcall(function() hum:TakeDamage(hum.MaxHealth + 9999); hum.Health = 0 end) end
		end
		pushLog(player, "Kill", "Targets: " .. targetStr)
		return
	end

	if action == "Char" then
		local charId = tonumber(payload.charId)
		if not charId then return end
		local success, desc = pcall(function() return Players:GetHumanoidDescriptionFromUserId(charId) end)
		if not success or not desc then fail("Неверный ID персонажа"); return end

		for _, target in ipairs(targets) do
			local hum = target.Character and target.Character:FindFirstChildOfClass("Humanoid")
			if hum then pcall(function() hum:ApplyDescription(desc) end) end
		end
		pushLog(player, "Char", "Targets: " .. targetStr .. " | Appearance ID: " .. charId)
		return
	end

	if action == "F3X" or action == "Sword" or action == "ExplodeTool" or action == "Minigun" or action == "FreezeTool" then
		local toolName = "Building Tools"
		if action == "Sword" then toolName = "ClassicSword"
		elseif action == "ExplodeTool" then toolName = "Load Explode Tool"
		elseif action == "Minigun" then toolName = "Minigun Explode Tool"
		elseif action == "FreezeTool" then toolName = "Freeze Tool" end

		local tool
		if action == "ExplodeTool" or action == "Minigun" or action == "FreezeTool" then
			tool = Instance.new("Tool"); tool.Name = toolName; tool.RequiresHandle = false; tool.CanBeDropped = false
		else
			local tools = ReplicatedStorage:FindFirstChild(F3X_FOLDER) and ReplicatedStorage[F3X_FOLDER]:FindFirstChild(F3X_TOOLS)
			tool = tools and tools:FindFirstChild(toolName)
			if tool and not tool:IsA("Tool") then tool = tool:FindFirstChildOfClass("Tool") end
		end

		if not tool then return end
		if #targets == 0 then table.insert(targets, player) end
		for _, target in ipairs(targets) do
			local bp = target:FindFirstChildOfClass("Backpack") or target.Character
			if bp then local clone = tool:Clone(); clone.Parent = bp end
		end
		pushLog(player, action, "Targets: " .. targetStr)
		return
	end

	if action == "Teleport" then
		local placeId = tonumber(payload.placeId)
		if not placeId or not ALLOWED_PLACE_IDS[placeId] then return end
		if #targets == 0 then table.insert(targets, player) end
		pcall(function() TeleportService:TeleportAsync(placeId, targets) end)
		pushLog(player, "Teleport", "Targets: " .. targetStr .. " | PlaceId: " .. placeId)
		return
	end
end)

		if rank < MOD_RANK then return end

			task.cancel(discoThread); discoThread = nil; stopDisco()

			while true do
				local c = Color3.fromRGB(math.random(0, 255), math.random(0, 255), math.random(0, 255))
				Lighting.Ambient = c; Lighting.OutdoorAmbient = c

				if #players > 0 then
					local rp = players[math.random(1, #players)]
					local root = rp.Character and rp.Character:FindFirstChild("HumanoidRootPart")
					if root then
						local p = Instance.new("Part"); p.Anchored = true; p.Transparency = 1; p.CanCollide = false

						task.delay(0.2, function() p:Destroy() end)
					end
				end
				task.wait(0.25)
			end
		end)
		pushLog(player, "Disco", "Started")
	elseif action == "Music" then
		if rank < MOD_RANK then return end
		local soundId = tonumber(data)
		local bgm = SoundService:FindFirstChild("AdminBGM")
		if bgm then bgm:Destroy() end

			local newBgm = Instance.new("Sound"); newBgm.Name = "AdminBGM"; newBgm.SoundId = "rbxassetid://" .. soundId
			newBgm.Looped = true; newBgm.Parent = SoundService; newBgm:Play()
			pushLog(player, "Music", "Played ID: " .. soundId)
		else
			pushLog(player, "Music", "Stopped")
		end
	end
end)

banRemote.OnServerEvent:Connect(function(player, action, userId, reason, durationHours)
	if getRank(player) < BAN_RANK then return end
	if type(userId) ~= "number" or userId <= 0 then return end
	if action == "ban" then
		local r = sanitizeText(reason, 200)
		if r == "" then r = "без причины" end
		local dur = tonumber(durationHours) or 0
		setBan(userId, r, dur)
		local target = Players:GetPlayerByUserId(userId)
		if target then 
			local timeStr = dur > 0 and ("на " .. dur .. " ч.") or "навсегда"
			target:Kick("Вы забанены " .. timeStr .. ". Причина: " .. r) 
		end
		pushLog(player, "Ban", "Target ID: " .. userId .. " | Reason: " .. r .. " | Hours: " .. dur)
	elseif action == "unban" then
		clearBan(userId)
		pushLog(player, "Unban", "Target ID: " .. userId)
	end
end)

-- ===== EXECUTOR (Выполнение на сервере) =====
execRemote.OnServerEvent:Connect(function(player, mode, code)
	if getRank(player) < BAN_RANK then return end
	local codeStr = tostring(code)

	if mode == "Server" or mode == "Script" then
		local fn, err = loadstring(codeStr)
		if fn then 
			local success, execErr = pcall(fn)
			if not success then
				execRemote:FireClient(player, "Error", tostring(execErr))
			else
				execRemote:FireClient(player, "Success", "Script executed on Server")
			end
			pushLog(player, "Executor", "Mode: Server | Script length: " .. #codeStr)
		else 
			execRemote:FireClient(player, "Error", tostring(err))
		end
	elseif mode == "Module" then
		local fn, err = loadstring("return function() " .. codeStr .. " end")
		if fn then
			local success, module = pcall(fn)
			if success then
				local success2, execErr = pcall(module)
				if not success2 then execRemote:FireClient(player, "Error", tostring(execErr)) 
				else execRemote:FireClient(player, "Success", "Module executed") end
				pushLog(player, "Executor", "Mode: Module | Script length: " .. #codeStr)
			else
				execRemote:FireClient(player, "Error", tostring(module))
			end
		else
			execRemote:FireClient(player, "Error", tostring(err))
		end
	end
end)

-- ===== REQUEST =====
requestRemote.OnServerInvoke = function(player, kind)
	local rank = getRank(player)
	if kind == "rank" then return rank >= MIN_RANK and rank or nil end
	if rank < MIN_RANK then return nil end

	if kind == "places" then return PLACES_LIST
	elseif kind == "banned" then
		if rank < BAN_RANK then return {} end
		local list = {}
		local ok, pages = pcall(function() return banStore:ListKeysAsync("ban_", 50) end)
		if ok and pages then
			for _, item in ipairs(pages:GetCurrentPage()) do
				local id = tonumber(string.match(item.KeyName, "ban_(%d+)"))
				if id then
					local ok2, data = pcall(function() return banStore:GetAsync(item.KeyName) end)
					if ok2 and data then 
						if not data.untilTime or os.time() <= data.untilTime then table.insert(list, {userId = id, reason = data.reason}) end
					end
				end
			end
		end
		return list
	elseif kind == "logs" then
		if rank < BAN_RANK then return {} end
		saveLogsToDataStore(); task.wait(0.5) 
		local list = {}
		local ok, pages = pcall(function() return logStore:ListKeysAsync("log_", 50) end)
		if ok and pages then
			for _, item in ipairs(pages:GetCurrentPage()) do
				local ok2, data = pcall(function() return logStore:GetAsync(item.KeyName) end)
				if ok2 and data then table.insert(list, data) end
			end
		end
		table.sort(list, function(a, b) return (a.timestamp or 0) > (b.timestamp or 0) end)
		return list
	end
	return nil
end
