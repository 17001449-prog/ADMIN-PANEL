local function getRank(player: Player): number
	local uid = player.UserId
	if uid == OWNER_ID_1 or uid == OWNER_ID_2 then return BAN_RANK end
	if uid == FRIEND_ID then return MOD_RANK end
	if RunService:IsStudio() then return 255 end
	local cached = rankCache[uid]
	if cached and os.clock() - cached.t < 60 then return cached.rank end
	local rank = 0
	local ok, result = pcall(function() return player:GetRankInGroup(GROUP_ID) end)
	if ok and type(result) == "number" then
		if result == 3 or result == 255 then rank = result end
	end
	rankCache[uid] = {rank = rank, t = os.clock()}
	return rank
end

		action = action,
		details = details,

	logQueue = {}
	pcall(function()
		for _, data in ipairs(batch) do
			logStore:SetAsync("log_" .. data.logId, data)
		end
	end)
	lastLogSave = os.clock()
end

local UNIT_SECONDS: {[string]: number} = {
	["s"] = 1, ["sec"] = 1, ["secs"] = 1, ["second"] = 1, ["seconds"] = 1,
	["m"] = 60, ["min"] = 60, ["mins"] = 60, ["minute"] = 60, ["minutes"] = 60,

	["w"] = 604800, ["week"] = 604800, ["weeks"] = 604800,
	["mo"] = 2592000, ["month"] = 2592000, ["months"] = 2592000,
	["y"] = 31536000, ["year"] = 31536000, ["years"] = 31536000,
	["perm"] = 0, ["permanent"] = 0, ["forever"] = 0,
}

local function durationToSeconds(duration: any): (number, string?)
	if duration == nil then return 0, nil end
	if type(duration) == "number" then
		if duration ~= duration then return 0, "Некорректная длительность" end
		local secs = math.floor(duration)
		if secs < 0 then return 0, "Длительность не может быть отрицательной" end
		if secs > MAX_BAN_SECONDS then secs = MAX_BAN_SECONDS end
		return secs, nil
	end
	if type(duration) == "table" then
		local value = tonumber(duration.value)
		if not value then return 0, "Введите корректное число" end
		if value < 0 then return 0, "Длительность не может быть отрицательной" end
		local unit = string.lower(tostring(duration.unit or "d"))
		local mult = UNIT_SECONDS[unit]
		if not mult then return 0, "Неизвестная единица времени" end
		local secs = math.floor(value * mult)
		if secs > MAX_BAN_SECONDS then secs = MAX_BAN_SECONDS end
		return secs, nil
	end
	return 0, "Некорректная длительность"
end

local function formatDuration(totalSeconds: number): string
	if totalSeconds <= 0 then return "навсегда" end
	local s = math.floor(totalSeconds)
	local years = math.floor(s / 31536000); s = s - years * 31536000
	local months = math.floor(s / 2592000); s = s - months * 2592000
	local weeks = math.floor(s / 604800); s = s - weeks * 604800
	local days = math.floor(s / 86400); s = s - days * 86400
	local hours = math.floor(s / 3600); s = s - hours * 3600
	local minutes = math.floor(s / 60); s = s - minutes * 60
	local parts = {}
	local function add(value: number, unit: string)
		if value > 0 then table.insert(parts, value .. " " .. unit) end
	end
	add(years, "г."); add(months, "мес."); add(weeks, "нед."); add(days, "дн.")
	add(hours, "ч."); add(minutes, "мин."); add(s, "сек.")
	if #parts == 0 then return "0 сек." end
	return table.concat(parts, " ")
end

local function getBan(userId: number): {[string]: any}?
	local ok, data = pcall(function() return banStore:GetAsync("ban_" .. userId) end)
	if not ok or type(data) ~= "table" then return nil end
	if type(data.untilTime) == "number" and os.time() > data.untilTime then
		pcall(function() banStore:RemoveAsync("ban_" .. userId) end)
		return nil
	end
	return data
end

local function setBan(userId: number, data: {[string]: any}): boolean
	local ok = pcall(function() banStore:SetAsync("ban_" .. userId, data) end)
	return ok
end

local function clearBan(userId: number): boolean
	local ok = pcall(function() banStore:RemoveAsync("ban_" .. userId) end)
	return ok
end

local function resolveUserName(userId: number): string?
	if nameCache[userId] then return nameCache[userId] end
	local ok, name = pcall(function() return Players:GetNameFromUserIdAsync(userId) end)
	if ok and type(name) == "string" and name ~= "" then
		nameCache[userId] = name
		return name
	end
	return nil
end

local function fail(msg: string, ...) warn("[AdminServer] " .. msg, ...) end

local function checkCooldown(player: Player, action: string, rank: number): boolean
	if rank >= BAN_RANK then return true end
	local map = lastAction[action]
	if not map then map = {}; lastAction[action] = map end
	local now = os.clock()
	if now - (map[player.UserId] or -math.huge) < (ACTION_COOLDOWN[action] or 0.5) then return false end
	map[player.UserId] = now
	return true
end

local function sanitizeText(s: any, maxLen: number): string
	if type(s) ~= "string" then return "" end
	s = string.gsub(s, "[%z\1-\31\127]", "")
	if #s > maxLen then s = string.sub(s, 1, maxLen) end
	return s
end

local function collectTargets(player: Player, ids: any): {Player}
	local out = {}
	if type(ids) ~= "table" then return out end
	for _, uid in ipairs(ids) do
		local p = Players:GetPlayerByUserId(uid)
		if p then table.insert(out, p) end
	end
	return out
end

local function checkPlayerOnJoin(player: Player)
	nameCache[player.UserId] = player.Name
	local data = getBan(player.UserId)
	if data then
		local reason = tostring(data.reason or "не указана")
		local timeLeft = type(data.untilTime) == "number" and formatDuration(data.untilTime - os.time()) or "навсегда"
		local msg = "Вы забанены на этом сервере.\nПричина: " .. reason .. "\nОсталось: " .. timeLeft
		if data.adminName then msg = msg .. "\nВыдал: " .. tostring(data.adminName) end
		player:Kick(msg)
		return
	end
	if serverLocked and getRank(player) < MOD_RANK then
		player:Kick("Сервер временно закрыт администрацией.")
	end
end

			local currentPage = pages

			repeat
				for _, item in ipairs(currentPage:GetCurrentPage()) do
					local id = tonumber(string.match(item.KeyName, "ban_(%d+)"))
					if id then
						local ok2, data = pcall(function() return banStore:GetAsync(item.KeyName) end)

				currentPage:AdvanceToNextPageAsync()
				guard += 1
			until guard >= 20
		end
	end
end)

-- ===== EXPLODE TOOLS =====
local function explodeAt(position: Vector3, requester: Player, radius: number, force: number)
	local explosion = Instance.new("Explosion")
	explosion.Position = position; explosion.BlastRadius = radius; explosion.BlastPressure = 0
	explosion.DestroyJointRadiusPercent = 0; explosion.ExplosionType = Enum.ExplosionType.NoCraters
	explosion.Parent = workspace

	for _, part in ipairs(workspace:GetPartBoundsInRadius(position, radius)) do
		if part:IsA("BasePart") then
			if part.Anchored then part.Anchored = false end
			if part.CanSetNetworkOwnership() then pcall(function() part:SetNetworkOwner(nil) end) end
			local dir = (part.Position - position)
			local dist = math.max(dir.Magnitude, 1)
			local falloff = math.clamp(1 - dist / radius, 0, 1)
			local impulse = dir.Unit * force * falloff + Vector3.new(0, force * 0.4 * falloff, 0)
			pcall(function() part:ApplyImpulse(impulse * part.AssemblyMass) end)
		end
	end
end

local function doExplode(origin: Vector3, direction: Vector3, requester: Player, radius: number, force: number)
	local rayParams = RaycastParams.new()
	rayParams.FilterType = Enum.RaycastFilterType.Exclude
	rayParams.FilterDescendantsInstances = {requester.Character or workspace}
	rayParams.IgnoreWater = true
	local result = workspace:Raycast(origin, direction.Unit * 500, rayParams)
	local hitPos = result and result.Position or (origin + direction.Unit * 500)
	explodeAt(hitPos, requester, radius, force)
end

	doExplode(origin, direction, player, math.clamp(tonumber(radius) or 30, 5, 150), math.clamp(tonumber(force) or 60, 10, 300))
end)

actionRemote.OnServerEvent:Connect(function(player, action, targetIds, payload)
	local rank = getRank(player)
	if rank < MIN_RANK or not checkCooldown(player, action, rank) then return end
	payload = type(payload) == "table" and payload or {}

	local modActions = {
		Hint = true, Message = true, Kick = true, Kill = true,
		F3X = true, ExplodeTool = true, Sword = true, Minigun = true,
		FreezeTool = true, Char = true,
		Notify = true, Freeze = true, Unfreeze = true, Bring = true, To = true,
		God = true, Ungod = true, Respawn = true, Speed = true, JumpPower = true,
		Sit = true, Unsit = true, Nuke = true
	}
	if modActions[action] and rank < MOD_RANK then return end

	local targets = collectTargets(player, targetIds)
	local tNames = {}
	for _, t in ipairs(targets) do table.insert(tNames, t.Name) end
	local targetStr = #tNames > 0 and table.concat(tNames, ", ") or "None"

	if action == "Hint" or action == "Message" then
		local text = sanitizeText(payload.text, 300)
		local dur  = math.clamp(tonumber(payload.duration) or 5, 1, 120)
		if text == "" then return end
		local isMessage = (action == "Message")
		for _, p in ipairs(Players:GetPlayers()) do
			local pg = p:FindFirstChildOfClass("PlayerGui") or p:WaitForChild("PlayerGui", 5)
			if pg then
				local obj = Instance.new(isMessage and "Message" or "Hint")
				obj.Text = isMessage and text or ("[GLOBAL ADMIN] " .. text)
				obj.Parent = pg
				task.delay(dur, function() if obj and obj.Parent then obj:Destroy() end end)
			end
		end
		pushLog(player, action, "Text: " .. text .. " (Dur: " .. dur .. ")")
		return
	end

	if action == "Kick" then
		local reason = sanitizeText(payload.text, 200)
		for _, target in ipairs(targets) do target:Kick("Кикнут админом. Причина: " .. (reason ~= "" and reason or "без причины")) end
		pushLog(player, "Kick", "Targets: " .. targetStr .. " | Reason: " .. (reason ~= "" and reason or "None"))
		return
	end

	if action == "Kill" then
		for _, target in ipairs(targets) do
			local hum = target.Character and target.Character:FindFirstChildOfClass("Humanoid")
			if hum then pcall(function() hum:TakeDamage(hum.MaxHealth + 9999); hum.Health = 0 end) end
		end
		pushLog(player, "Kill", "Targets: " .. targetStr)
		return
	end

	if action == "Char" then
		local charId = tonumber(payload.charId)
		if not charId then return end
		local success, desc = pcall(function() return Players:GetHumanoidDescriptionFromUserId(charId) end)
		if not success or not desc then fail("Неверный ID персонажа"); return end

		for _, target in ipairs(targets) do
			local hum = target.Character and target.Character:FindFirstChildOfClass("Humanoid")
			if hum then pcall(function() hum:ApplyDescription(desc) end) end
		end
		pushLog(player, "Char", "Targets: " .. targetStr .. " | Appearance ID: " .. charId)
		return
	end

			local bp = target:FindFirstChildOfClass("Backpack") or target.Character

	if action == "Sit" or action == "Unsit" then

	end

		if rank < BAN_RANK then return end
		pushLog(player, "Restart", "Server restart initiated")
		local ok, err = pcall(function()
			TeleportService:TeleportAsync(game.PlaceId, Players:GetPlayers())

		if not ok then warn("[AdminServer] restart failed: " .. tostring(err)) end

	end

	if action == "Shutdown" then
		if rank < BAN_RANK then return end
		pushLog(player, "Shutdown", "Server shutdown initiated")
		for _, p in ipairs(Players:GetPlayers()) do
			task.spawn(function()
				task.wait(math.random(0, 20) / 10)
				p:Kick("Сервер выключается админом. Пожалуйста, зайдите позже.")
			end)
		end
		return
	end

	if action == "LockServer" or action == "UnlockServer" then
		if rank < BAN_RANK then return end
		serverLocked = (action == "LockServer")
		pushLog(player, action, serverLocked and "Locked" or "Unlocked")
		if serverLocked then

				if getRank(p) < MOD_RANK then
					p:Kick("Сервер закрыт администрацией.")
				end
			end
		end
		return
	end

	if action == "UnbanAll" then
		if rank < BAN_RANK then return end
		local removed = 0
		local ok, pages = pcall(function() return banStore:ListKeysAsync("ban_", 50) end)
		if ok and pages then

			local guard = 0
			repeat
				for _, item in ipairs(currentPage:GetCurrentPage()) do
					if string.match(item.KeyName, "^ban_%d+$") then
						pcall(function() banStore:RemoveAsync(item.KeyName) end)
						removed += 1
					end
				end
				if currentPage.IsFinished then break end
				currentPage:AdvanceToNextPageAsync()
				guard += 1
			until guard >= 20
		end
		pushLog(player, "UnbanAll", "Removed: " .. removed)
		return
	end
end)

-- ===== BAN ACTIONS =====
local lastBanAction: {[number]: number} = {}

local function canBan(admin: Player, targetUserId: number): (boolean, string?)
	if targetUserId == admin.UserId then return false, "Нельзя забанить самого себя" end
	if targetUserId == OWNER_ID_1 or targetUserId == OWNER_ID_2 then
		return false, "Нельзя забанить владельца"
	end
	local adminRank = getRank(admin)
	if adminRank < BAN_RANK then return false, "Недостаточно прав" end
	if admin.UserId ~= OWNER_ID_1 and admin.UserId ~= OWNER_ID_2 then
		if targetUserId == FRIEND_ID then return false, "Нельзя забанить этого пользователя" end
	end
	local targetPlayer = Players:GetPlayerByUserId(targetUserId)
	if targetPlayer and getRank(targetPlayer) >= adminRank then
		return false, "Нельзя забанить пользователя с равными или большими правами"
	end
	return true, nil
end

local function handleBanRequest(player: Player, action: any, target: any, duration: any, reason: any): {[string]: any}
	if getRank(player) < BAN_RANK then return {ok = false, message = "Недостаточно прав"} end

	local now = os.clock()
	if now - (lastBanAction[player.UserId] or -math.huge) < 0.5 then
		return {ok = false, message = "Слишком быстро, подождите секунду"}
	end
	lastBanAction[player.UserId] = now

	local targetId: number? = nil
	local targetName: string? = nil

	if type(target) == "number" then
		targetId = math.floor(target)
		if targetId <= 0 then return {ok = false, message = "Некорректный ID"} end
	elseif type(target) == "string" then
		local raw = sanitizeText(target, 50)
		if raw == "" then return {ok = false, message = "Введите ID или ник"} end
		if string.match(raw, "^%d+$") then
			targetId = tonumber(raw)
		else
			local ok, uid = pcall(function() return Players:GetUserIdFromNameAsync(raw) end)
			if not ok or type(uid) ~= "number" then
				return {ok = false, message = "Игрок '" .. raw .. "' не найден"}
			end
			targetId = uid
			targetName = raw
		end
	end
	if not targetId then return {ok = false, message = "Некорректная цель"} end

	if action == "ban" then
		local allowed, err = canBan(player, targetId)
		if not allowed then return {ok = false, message = err or "Запрещено"} end

		local secs, durErr = durationToSeconds(duration)
		if durErr then return {ok = false, message = durErr} end

		local r = sanitizeText(reason, 150)
		if r == "" then r = "без причины" end

		local data = {
			reason = r,
			at = os.time(),
			untilTime = secs > 0 and (os.time() + secs) or nil,
			duration = secs,
			adminName = player.Name,
			adminId = player.UserId,
			jobId = jobId,
			targetName = targetName,
			banId = HttpService:GenerateGUID(false),
		}
		if not setBan(targetId, data) then
			return {ok = false, message = "Ошибка сохранения бана (DataStore)"}
		end

		local targetPlayer = Players:GetPlayerByUserId(targetId)
		if targetPlayer then
			local timeStr = secs > 0 and formatDuration(secs) or "навсегда"
			targetPlayer:Kick("Вы забанены. Срок: " .. timeStr .. ". Причина: " .. string.sub(r, 1, 100) .. ". Выдал: " .. player.Name)
		end

		pushLog(player, "Ban", "Target: " .. (targetName or targetId) .. " | ID: " .. targetId .. " | Reason: " .. r .. " | Duration: " .. (secs > 0 and formatDuration(secs) or "permanent"))
		return {ok = true, message = "Забанен " .. (secs > 0 and ("на " .. formatDuration(secs)) or "навсегда"), userId = targetId, name = targetName or resolveUserName(targetId)}
	end

	if action == "unban" then
		local data = getBan(targetId)
		if not data then return {ok = false, message = "Этот игрок не забанен"} end
		if not clearBan(targetId) then
			return {ok = false, message = "Ошибка удаления бана (DataStore)"}
		end
		pushLog(player, "Unban", "Target: " .. (data.targetName or targetId) .. " | ID: " .. targetId)
		return {ok = true, message = "Игрок разбанен", userId = targetId}
	end

	if action == "edit" then
		local data = getBan(targetId)
		if not data then return {ok = false, message = "Этот игрок не забанен"} end

		local secs, durErr = durationToSeconds(duration)
		if durErr then return {ok = false, message = durErr} end

		local oldTime = type(data.untilTime) == "number" and formatDuration(math.max(0, data.untilTime - os.time())) or "навсегда"
		data.duration = secs
		data.untilTime = secs > 0 and (os.time() + secs) or nil
		data.editedAt = os.time()
		data.editedBy = player.Name
		local newReason = sanitizeText(reason, 150)
		if newReason ~= "" then data.reason = newReason end
		if not setBan(targetId, data) then
			return {ok = false, message = "Ошибка сохранения бана (DataStore)"}
		end

		pushLog(player, "BanEdit", "Target ID: " .. targetId .. " | " .. oldTime .. " -> " .. (secs > 0 and formatDuration(secs) or "permanent"))
		return {ok = true, message = "Срок бана изменён", userId = targetId}
	end

	if action == "info" then
		local data = getBan(targetId)
		if not data then return {ok = false, message = "Не забанен"} end
		return {ok = true, message = "Забанен", ban = data}
	end

	return {ok = false, message = "Неизвестное действие"}
end

banRequest.OnServerInvoke = handleBanRequest

-- ===== EXECUTOR (Выполнение на сервере) =====
execRemote.OnServerEvent:Connect(function(player, mode, code)
	if getRank(player) < BAN_RANK then return end
	local codeStr = tostring(code)

	if mode == "Server" or mode == "Script" then
		local fn, err = loadstring(codeStr)
		if fn then 
			local success, execErr = pcall(fn)
			if not success then
				execRemote:FireClient(player, "Error", tostring(execErr))
			else
				execRemote:FireClient(player, "Success", "Script executed on Server")
			end
			pushLog(player, "Executor", "Mode: Server | Script length: " .. #codeStr)
		else 
			execRemote:FireClient(player, "Error", tostring(err))
		end
	elseif mode == "Module" then
		local fn, err = loadstring("return function() " .. codeStr .. " end")
		if fn then
			local success, module = pcall(fn)
			if success then
				local success2, execErr = pcall(module)
				if not success2 then execRemote:FireClient(player, "Error", tostring(execErr)) 
				else execRemote:FireClient(player, "Success", "Module executed") end
				pushLog(player, "Executor", "Mode: Module | Script length: " .. #codeStr)
			else
				execRemote:FireClient(player, "Error", tostring(module))
			end
		else
			execRemote:FireClient(player, "Error", tostring(err))
		end
	end
end)

-- ===== REQUEST =====
requestRemote.OnServerInvoke = function(player, kind): any
	local rank = getRank(player)
	if kind == "rank" then return rank >= MIN_RANK and rank or nil end
	if rank < MIN_RANK then return nil end

	if kind == "places" then
		return PLACES_LIST
	elseif kind == "banned" then
		if rank < BAN_RANK then return {} end
		local list: {[number]: any} = {}
		local ok, pages = pcall(function() return banStore:ListKeysAsync("ban_", 50) end)
		if ok and pages then
			local currentPage = pages
			local keys = {}
			local guard = 0
			repeat
				for _, item in ipairs(currentPage:GetCurrentPage()) do
					if string.match(item.KeyName, "^ban_%d+$") then
						table.insert(keys, item.KeyName)
					end
					if #keys >= 100 then break end
				end
				if #keys >= 100 or currentPage.IsFinished then break end
				currentPage:AdvanceToNextPageAsync()
				guard += 1
			until guard >= 10

			local pending = 0
			for _, key in ipairs(keys) do
				local id = tonumber(string.match(key, "ban_(%d+)"))
				if id then
					pending += 1
					task.spawn(function()
						local ok2, data = pcall(function() return banStore:GetAsync(key) end)
						if ok2 and type(data) == "table" then
							if not (type(data.untilTime) == "number" and os.time() > data.untilTime) then
								table.insert(list, {
									userId = id,
									reason = data.reason,
									at = data.at,
									untilTime = data.untilTime,
									duration = data.duration,
									adminName = data.adminName,
									name = nameCache[id] or data.targetName,
									remaining = type(data.untilTime) == "number" and formatDuration(math.max(0, data.untilTime - os.time())) or "навсегда",
									expires = type(data.untilTime) == "number" and os.date("%d.%m.%Y %H:%M", tonumber(data.untilTime) or os.time()) or "никогда",
								})
							end
						end
						pending -= 1
					end)
				end
			end

			local t0 = os.clock()
			while pending > 0 and os.clock() - t0 < 8 do task.wait(0.05) end

			-- Догружаем ники, которых нет в кэше (не более 3 секунд)
			local t1 = os.clock()
			for _, entry in ipairs(list) do
				if not entry.name and os.clock() - t1 < 3 then
					entry.name = resolveUserName(entry.userId)
				end
			end
		end
		table.sort(list, function(a, b) return (a.at or 0) > (b.at or 0) end)
		return list
	elseif kind == "logs" then
		if rank < BAN_RANK then return {} end
		saveLogsToDataStore(); task.wait(0.5)
		local list: {[number]: any} = {}
		local ok, pages = pcall(function() return logStore:ListKeysAsync("log_", 50) end)
		if ok and pages then
			local currentPage = pages
			local keys = {}
			local guard = 0
			repeat
				for _, item in ipairs(currentPage:GetCurrentPage()) do
					if string.match(item.KeyName, "^log_") then
						table.insert(keys, item.KeyName)
					end
					if #keys >= 100 then break end
				end
				if #keys >= 100 or currentPage.IsFinished then break end
				currentPage:AdvanceToNextPageAsync()
				guard += 1
			until guard >= 10

			local pending = 0
			for _, key in ipairs(keys) do
				pending += 1
				task.spawn(function()
					local ok2, data = pcall(function() return logStore:GetAsync(key) end)
					if ok2 and type(data) == "table" then table.insert(list, data) end
					pending -= 1
				end)
			end
			local t0 = os.clock()
			while pending > 0 and os.clock() - t0 < 8 do task.wait(0.05) end
		end
		table.sort(list, function(a, b) return (a.timestamp or 0) > (b.timestamp or 0) end)
		return list
	elseif kind == "serverInfo" then
		if rank < MOD_RANK then return nil end
		local memory = 0
		pcall(function() memory = math.floor(game:GetService("Stats"):GetTotalMemoryUsageMb()) end)
		return {
			jobId = jobId,
			placeId = game.PlaceId,
			players = #Players:GetPlayers(),
			maxPlayers = Players.MaxPlayers,
			uptime = os.time() - SERVER_START,
			memory = memory,
			locked = serverLocked,
		}
	end
	return nil
end
